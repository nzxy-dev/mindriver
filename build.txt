:build_msvc
echo [build] compiling mindriver with msvc (Release)...
if not exist "bin\out\windows" mkdir "bin\out\windows"
if not exist "bin\00Temp" mkdir "bin\00Temp"

set VCVARS="C:\Program Files (x86)\Microsoft Visual Studio\18\BuildTools\VC\Auxiliary\Build\vcvars64.bat"
if exist %VCVARS% call %VCVARS%

set RCEXE="C:\Program Files (x86)\Windows Kits\10\bin\10.0.26100.0\x64\rc.exe"
if not exist %RCEXE% set RCEXE=rc.exe


%RCEXE% /fo bin\00Temp\mindriver.res windows\mindriver.rc

cl.exe /nologo /O2 /DNDEBUG /D_CRT_SECURE_NO_WARNINGS /W3 /MD /I. /Iwindows /c main.c /Fo:bin\00Temp\main.obj
cl.exe /nologo /O2 /DNDEBUG /D_CRT_SECURE_NO_WARNINGS /W3 /MD /I. /Iwindows /c windows\initcore.c /Fo:bin\00Temp\initcore.obj
cl.exe /nologo /O2 /DNDEBUG /D_CRT_SECURE_NO_WARNINGS /W3 /MD /I. /Iwindows /c windows\resolver.c /Fo:bin\00Temp\resolver.obj
cl.exe /nologo /O2 /DNDEBUG /D_CRT_SECURE_NO_WARNINGS /W3 /MD /I. /Iwindows /c windows\servicefunc.c /Fo:bin\00Temp\servicefunc.obj

link.exe /NOLOGO /OUT:bin\out\windows\mindriver.exe bin\00Temp\main.obj bin\00Temp\initcore.obj bin\00Temp\resolver.obj bin\00Temp\servicefunc.obj bin\00Temp\mindriver.res ole32.lib user32.lib mfplat.lib mfreadwrite.lib mfuuid.lib avrt.lib

if errorlevel 1 goto :build_failed
echo [build] SUCCESS: bin\out\windows\mindriver.exe
goto :end
